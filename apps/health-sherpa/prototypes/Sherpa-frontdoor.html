        <p id="encryption-text" style="font-size: 22px; color: var(--text-secondary); opacity: 0; transition: opacity 0.5s ease-in-out; text-align: center; padding: 0 20px;"></p>

    <script>

    document.addEventListener('DOMContentLoaded', () => {

        // --- CONFIGURATION ---

        const PROMPTS = [

            "What's been on your mind about health? Let's break it down together...",

            "What's your main goal with this?",

            "What's your biggest concern or worry?",

            "What's your role in this situation?",

            "How familiar are you with this topic?"

        ];

        const LABELS = ['Question', 'Goal', 'Concern', 'Role', 'Expertise'];


        // --- STATE MANAGEMENT ---

        const state = {

            currentStep: 0,

            responses: Array(PROMPTS.length).fill(''),

            editingStep: -1,

            isComplete: false,

            isAnimating: false

        };


        // --- DOM ELEMENT CACHE ---

        const elements = {

            searchForm: document.getElementById('searchForm'),

            searchInput: document.getElementById('searchInput'),

            searchSubmit: document.getElementById('searchSubmit'),

            pillsContainer: document.getElementById('pillsContainer'),

            startButton: document.getElementById('startButton'),

            sherpaRing: document.querySelector('.sherpa-ring'),

            ringSegments: document.querySelectorAll('.ring-segment'),

            sherpaLogo: document.getElementById('sherpaLogo'),

            ringContainer: document.getElementById('ringContainer'),

            searchBox: document.getElementById('searchBox'),

            sherpaTagline: document.getElementById('sherpaTagline'),

            sherpaContainer: document.getElementById('sherpaContainer'),

            encryptionContainer: document.getElementById('encryption-sequence-container'),

            encryptionText: document.getElementById('encryption-text')

        };


        // --- CORE LOGIC ---


        function handleSubmit(e) {

            e.preventDefault();

            const value = elements.searchInput.value.trim();

            if (!value || state.isAnimating) return;


            if (state.editingStep >= 0) {

                state.responses[state.editingStep] = value;

                state.editingStep = -1;

            } else {

                state.responses[state.currentStep] = value;

                state.currentStep++;

            }


            elements.searchInput.value = '';

            updateUI();

        }


        function editResponse(index) {

            if (state.isAnimating) return;

            state.editingStep = index;

            updateUI();

        }


        function handleStartSession() {

            if (!state.isComplete || state.isAnimating) return;

            packageDataForWebApp();

            startAnalyzingAnimation();

        }

        

        function packageDataForWebApp() {

            const packagedData = {};

            state.responses.forEach((response, index) => {

                const label = LABELS[index] || `Response ${index + 1}`;

                packagedData[label] = response;

            });

            console.log("Data packaged for web-app handoff:", packagedData);

            // In a real application, you would send this JSON to your web-app

            // e.g., via fetch() or another method.

        }


        // --- UI UPDATE FUNCTIONS ---


        function updateUI() {

            updatePills();

            updatePlaceholder();

            updateLogoVisibility();

            checkCompletion();

            handleInput();

        }


        function updatePlaceholder() {

            elements.searchInput.disabled = false;

            let currentPromptText = "";


            if (state.editingStep >= 0) {

                elements.searchInput.value = state.responses[state.editingStep];

                elements.searchInput.focus();

                currentPromptText = PROMPTS[state.editingStep]; 

            } else if (state.currentStep < PROMPTS.length) {

                currentPromptText = PROMPTS[state.currentStep];

                elements.searchBox.style.opacity = '1';

            } else {

                currentPromptText = "All set! Click 'Start Private Session' to continue.";

                elements.searchInput.disabled = true;

                elements.searchBox.style.opacity = '0'; // Fade out search box

            }


            // Update tagline with a fade effect if the text is changing

            if (elements.sherpaTagline.textContent !== currentPromptText) {

                elements.sherpaTagline.style.opacity = '0';

                setTimeout(() => {

                    elements.sherpaTagline.textContent = currentPromptText;

                    elements.sherpaTagline.style.opacity = '1';

                }, 250); 

            }

        }


        function updatePills() {

            while (elements.pillsContainer.firstChild) {

                elements.pillsContainer.removeChild(elements.pillsContainer.firstChild);

            }

              

            elements.ringSegments.forEach(seg => seg.classList.remove('active'));


            state.responses.forEach((response, i) => {

                if (response) {

                    elements.pillsContainer.appendChild(createPill(i, response));

                    activateSegment(i + 1);

                }

            });

        }


        function createPill(index, text) {

            const pill = document.createElement('button');

            pill.className = `answer-pill ${state.editingStep === index ? 'editing' : ''}`;

            pill.textContent = `${LABELS[index]}: ${text}`;

            pill.onclick = () => editResponse(index);

            return pill;

        }


        function activateSegment(segmentNumber) {

            const segment = elements.ringSegments[segmentNumber - 1];

            if (segment) segment.classList.add('active');

        }


        function updateLogoVisibility() {

            const hasStarted = state.currentStep > 0 || state.responses.some(r => r.length > 0);

            const isRingVisible = elements.ringContainer.classList.contains('visible');


            if (hasStarted && !isRingVisible) {

                elements.sherpaLogo.classList.add('hidden');

                elements.ringContainer.classList.add('visible');

            }

        }

          

        function handleInput() {

            const hasInput = elements.searchInput.value.trim().length > 0;

            elements.searchSubmit.classList.toggle('visible', hasInput);

        }


        function checkCompletion() {

            const isComplete = state.responses.every(r => r.length > 0);

              

            if (isComplete !== state.isComplete) {

                state.isComplete = isComplete;

                elements.startButton.classList.toggle('enabled', isComplete);

                elements.sherpaRing.classList.toggle('spinning', isComplete);

            }

        }


        // --- ANIMATION ---


        function startAnalyzingAnimation() {

            state.isAnimating = true;

            elements.startButton.classList.add('disappearing');

            elements.searchBox.classList.add('disappearing');

            elements.sherpaTagline.style.opacity = '0';

            elements.searchForm.style.pointerEvents = 'none';


            // Phase 1: Pills fly to center

            setTimeout(() => {

                const pills = document.querySelectorAll('.answer-pill');

                pills.forEach((pill, index) => {

                    const rect = pill.getBoundingClientRect();

                    pill.style.top = `${rect.top}px`;

                    pill.style.left = `${rect.left}px`;

                    pill.style.width = `${rect.width}px`;

                    pill.style.height = `${rect.height}px`;

                    setTimeout(() => pill.classList.add('flying'), index * 100);

                });


                // Phase 2: Ring accelerates

                setTimeout(() => {

                    elements.sherpaRing.classList.remove('spinning');

                    elements.sherpaRing.classList.add('analyzing');

                }, 300);


                // Phase 3: Transition to the text sequence

                setTimeout(startEncryptionSequence, 3500); // Wait for ring animation to finish

            }, 500);

        }


        async function startEncryptionSequence() {

            // Hide the main container

            elements.sherpaContainer.style.opacity = '0';

            elements.pillsContainer.style.opacity = '0';


            // Show the dedicated container for the encryption text

            elements.encryptionContainer.style.display = 'flex';

            

            const encryptionPrompts = [

                "Ok, encrypting this session.",

                "Whatever you share, or upload, will be de-identified.",

                "Just extra safety, so nothing can be traced to you."

            ];


            const wait = (ms) => new Promise(resolve => setTimeout(resolve, ms));


            // Initial pause before the first message

            await wait(1000);


            for (let i = 0; i < encryptionPrompts.length; i++) {

                // Set text and fade in

                elements.encryptionText.textContent = encryptionPrompts[i];

                elements.encryptionText.style.opacity = '1';


                // Wait for message to be displayed

                await wait(3000); 

                

                // Fade out before the next loop

                elements.encryptionText.style.opacity = '0';


                // Wait for fade-out to complete before the next loop starts

                await wait(500); 

            }

            

            console.log("Animation complete. Ready for next experience.");

        }

          

        // --- EVENT LISTENERS ---

        function init() {

            updateUI();

            elements.searchForm.addEventListener('submit', handleSubmit);

            elements.searchInput.addEventListener('input', handleInput);

            elements.searchInput.addEventListener('focus', updateLogoVisibility);

            elements.startButton.addEventListener('click', handleStartSession);

        }


        // --- INITIALIZATION ---

        init();

    });
